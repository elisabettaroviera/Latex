\begin{frame}{CpG and $\beta$-values: The Language of Epigenetic Data}
    \begin{itemize}[label=-]
    \item \textbf{CpG sites}: genomic positions where DNA methylation can occur.
    \item \textbf{$\beta$-value $\in [0,1]$}: proportion of methylation at each CpG.
    \item \textbf{DNA methylation} modulates \textbf{gene regulation}.
    \item In cancer, both \textbf{hypermethylation} and \textbf{hypomethylation} emerge.
\end{itemize}

\section{Data \& Computational Challenges}



\begin{frame}{Accessing and Understanding the Data}
    \begin{itemize}[label=-]
        \item GEO datasets: \textbf{heterogeneous formats} and \textbf{incomplete metadata}.
        \item \alert{Small $n$, huge $p$} (up to $\sim$800k CpG sites).
        \item Raw \textbf{IDAT files} vs.\ preprocessed \textbf{$\beta$-value matrices}.
        \item \textbf{Unknown or inconsistent preprocessing} in some datasets.
    \end{itemize}

\begin{frame}{Can I Merge the Datasets? NO}
    \begin{itemize}[label=-]
        \item Different array platforms (\textbf{HM450} vs.\ \textbf{EPIC}).
        \item \textbf{Heterogeneous} and partly \textbf{unknown preprocessing pipelines}.
        \item Direct merge would introduce \alert{strong batch and platform effects}.
        \item Decision: perform all analyses \textbf{within each dataset} (intra-dataset).
    \end{itemize}

\section{Feature Engineering \& Modeling}

\begin{frame}{Feature Selection in High Dimensions}

    \begin{itemize}[label=-]
        \item $p \gg n$ → \textbf{dimensionality reduction} and filtering are essential.
        \item Technical filters: cross-reactive probes, SNP probes, non-CpG probes, Illumina manifest.
        \item Evaluate and correct potential batch effects (e.g., \textbf{COMBAT}).
        \item Three families of feature selection strategies:
        \begin{itemize}[label=-]
            \item \textbf{Statistical}: $\Delta\beta$, variance filtering, stability measures.
            \item \textbf{Biological}: promoters, CpG islands, curated gene sets.
            \item \textbf{ML-based}: Lasso, decision trees, embedded model selection.
        \end{itemize}
    \end{itemize}





\end{frame}



\begin{frame}{ML in Non-Ideal Conditions}

    \begin{itemize}[label=-]
        \item \textbf{Train / validation / test} with extremely small $n$ (tens of samples).
        \item Generalising across datasets (A $\rightarrow$ B) is often \textbf{unreliable} due to batch and platform effects.
        \item \alert{\textbf{How many features} to select? \textbf{Which metrics} are stable in the $p \gg n$ regime?}
        \item \textbf{Overfitting}, noise, and limited interpretability of complex models.
        \item Deep Learning: training from scratch is \textbf{infeasible} $\rightarrow$ only \textbf{pre-trained / transfer learning} approaches are realistic.
    \end{itemize}

    \vspace{0.4cm}
    \centering
    \begin{tabular}{lcccc}

    }

\end{frame}





\section{Conclusions}

\begin{frame}{Take-home Message}
    \begin{itemize}[label=-]
        \item \textbf{Real-world bioinformatics pipeline}: from noisy GEO data and incomplete metadata 
              to a \textbf{clean, analysable} methylation matrix via file conversion, manifest alignment, 
              probe filtering, and batch assessment.
        \item \textbf{Epigenetics is deeply quantitative}: hundreds of thousands of CpGs, 
              \alert{$p \gg n$}, non-Gaussian $\beta$-distributions, strong batch and probe-design effects.
        \item \textbf{Role of mathematics and statistics}: exploratory analysis, 
              stability and variability measures, dimensionality reduction, feature selection, 
              and careful model evaluation under small-$n$/large-$p$ constraints.
        \item \textbf{Machine learning in non-ideal conditions}: limited data, strong regularisation, 
              preference for simpler and interpretable models; deep learning only via 
              \textbf{transfer learning} or as a complement, not as a starting point.
    \end{itemize}
    
    \vspace{0.25cm}
    {\small\color{gray}
    \textbf{Big picture}: working at the interface of \textbf{biology, statistics, and computing} 
              is challenging but extremely rewarding — and research projects like this are 
              \alert{much closer to our skill set than they might appear}.
    }
\end{frame}
